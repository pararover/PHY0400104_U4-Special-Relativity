\section{Effects of relativity}

	\subsection{Velocity transformation}
		Consider two inertial frames of reference $S$ and $S'$ with relative velocity $v$ along the $xx'$-axes.
		\begin{figure}[h!]
			\centering
			\includegraphics[width=0.45\textwidth]{figures/fig_velocity-transformation.png}
		\end{figure}
		Suppose a particle at $P$ is moving with a velocity $\vb{u}$ with respect to $S$ and with a velocity $\vb{u}'$ with respect to $S'$.
		
		Then the components of its velocity with respect to these two frames are as follows:
		\begin{align*}
			u_x=\dv{x}{t}\quad& u_y=\dv{y}{t}\quad u_z=\dv{z}{t} \\
			u_x'=\dv{x'}{t'}\quad& u_y'=\dv{y'}{t'}\quad u_z'=\dv{z'}{t'}
		\end{align*}
		
		Now, from the Lorentz transformation of coordinates from $S$ to $S'$, we have
		\begin{align*}
			x'&=\gamma\qty(x-\beta ct) \\
			\implies\dd{x'}&=\gamma\qty(\dd{x}-\beta c\dd{t}) \\
			\implies\dd{x'}&=\gamma\qty(\dd{x}-v\dd{t})
		\end{align*}
		Again
		\begin{align*}
			ct'&=\gamma\qty(ct-\beta x) \\
			\implies c\dd{t'}&=\gamma\qty(c\dd{t}-\beta\dd{x}) \\
			\implies\dd{t'}&=\gamma\qty(\dd{t}-\dfrac{\beta}{c}\dd{x})=\gamma\qty(\dd{t}-\dfrac{v/c}{c}\dd{x}) \\
			\implies\dd{t'}&=\gamma\qty(\dd{t}-\dfrac{v}{c^2}\dd{x})
		\end{align*}
		Finally,
		\begin{align*}
			\dd{y'}=\dd{y}\qq{and}\dd{z'}=\dd{z}
		\end{align*}
		
		Now, we obtain
		\begin{align*}
			u_x'=\dv{x'}{t'}&=\dfrac{\gamma\qty(\dd{x}-v\dd{t})}{\gamma\qty(\dd{t}-{v}/{c^2}\dd{x})} \\
				&=\dfrac{\dd{t}\qty(\dv*{x}{t}-v)}{\dd{t}\qty(1-v/c^2\cdot\dv*{x}{t})} \\
			\implies u_x'&=\dfrac{u_x-v}{1-u_xv/c^2}
		\end{align*}
		
		Likewise,
		\begin{align*}
			u_y'=\dv{y'}{t'}&=\dfrac{\dd{y}}{\gamma\qty(\dd{t}-{v}/{c^2}\dd{x})} \\
				&=\dfrac{\dv*{y}{t}}{\gamma\qty(\dd{t}-v/c^2\cdot\dv*{x}{t})} \\
			\implies u_y'&=\dfrac{u_y\sqrt{1-\beta^2}}{1-u_xv/c^2}
		\end{align*}
		
		Similarly,
		\begin{align*}
			u_z'&=\dfrac{u_z\sqrt{1-\beta^2}}{1-u_xv/c^2}
		\end{align*}
		
		Therefore, we obtain the \emph{Lorentz velocity transformation} equations as
		\begin{equation*}
			\boxed{
				\begin{aligned}
					u_x'&=\dfrac{u_x-v}{1-u_xv/c^2} \\
					u_y'&=\dfrac{u_y\sqrt{1-\beta^2}}{1-u_xv/c^2} \\
					u_z'&=\dfrac{u_z\sqrt{1-\beta^2}}{1-u_xv/c^2}
				\end{aligned}
			}
		\end{equation*}
		
		The \emph{inverse Lorentz velocity transformation} equations would then be
		\begin{equation*}
			\boxed{
				\begin{aligned}
					u_x&=\dfrac{u_x'+v}{1+u_x'v/c^2} \\
					u_y&=\dfrac{u_y'\sqrt{1-\beta^2}}{1+u_x'v/c^2} \\
					u_z&=\dfrac{u_z'\sqrt{1-\beta^2}}{1+u_x'v/c^2}
				\end{aligned}
			}
		\end{equation*}
		
		\textbf{Special case}: Suppose the motion of $P$ is along the $xx'$-axes, then we would have
		\begin{align*}
			u_x=u\quad u_y=0\quad u_z=0
		\end{align*}
		Then we would have $u_x'=u'$ and $u_y'=u_z'=0$. Then we obtain the forward and inverse Lorentz velocity transformations to be
		\begin{equation*}
			\boxed{
				u'=\dfrac{u-v}{1-uv/c^2}\qq{and}u=\dfrac{u'+v}{1+u'v/c^2}
			}
		\end{equation*}
		The above equations are referred to as \emph{Einstein's law of relativistic velocity addition}.
		
		\textbf{Example}: A spaceship leaves the Earth at a speed of $0.6c$. A rocket leaves the spaceship at a speed of $0.9c$ with respect to the spaceship. Calculate the speed of the rocket with respect to the Earth if
		\begin{enumerate}[i)]
			\item it is fired in the same direction as the spaceship.
			\item it is fired in the opposite direction as the spaceship.
		\end{enumerate}
		\textbf{Solution}: Here we consider
		\begin{align*}
			S&\text{: the Earth} \\
			S'&\text{: the spaceship} \\
			P&\text{: the rocket}
		\end{align*}
		Hence, we have
		\begin{align*}
			\text{speed of }S'\text{ w.r.t. }S&=\text{speed of the spaceship w.r.t. the Earth}=|v|=0.6c \\
			\text{speed of }P\text{ w.r.t. }S'&=\text{speed of the rocket w.r.t. the spaceship}=|u'|=0.9c \\
			\text{velocity of }P\text{ w.r.t. }S&=\text{speed of the rocket w.r.t. the Earth}=u=?
		\end{align*}
		\begin{enumerate}[i)]
			\item When the rocket is fired in the same direction as the spaceship
			\begin{figure}[h!]
				\centering
				\includegraphics[width=0.7\textwidth]{figures/fig_vel-addn-example-01.png}
			\end{figure}
			
			Here we have
			\begin{align*}
				v=0.6c,\quad u'=0.9c
			\end{align*}
			Then from Einstein's law of relativistic velocity addition, we have
			\begin{align*}
				u&=\dfrac{u'+v}{1+u'v/c^2} \\
					&=\dfrac{0.9c+0.6c}{1+(0.9c\times0.6c)/c^2} \\
					&=\dfrac{1.5c}{1+0.54c^2/c^2} \\
					&=\dfrac{1.5c}{1.54} \\
				\implies u&=0.974c
			\end{align*}
			
			\item When the rocket is fired in the opposite direction as the spaceship
			\begin{figure}[h!]
				\centering
				\includegraphics[width=0.7\textwidth]{figures/fig_vel-addn-example-02.png}
			\end{figure}
			
			Here we have
			\begin{align*}
				v=0.6c,\quad u'=-0.9c
			\end{align*}
			Then from Einstein's law of relativistic velocity addition, we have
			\begin{align*}
				u&=\dfrac{u'+v}{1+u'v/c^2} \\
				&=\dfrac{-0.9c+0.6c}{1+(-0.9c\times0.6c)/c^2} \\
				&=\dfrac{-0.3c}{1-0.54c^2/c^2} \\
				&=\dfrac{-0.3c}{0.46} \\
				\implies u&=-0.652c
			\end{align*}
			Here the $-$ sign indicates that the rocket is moving towards the Earth.
		\end{enumerate}
		
	\subsection{Length contraction}
		\setcounter{equation}{0}
		Suppose we have a rod of natural length $L_0$, whose ends are marked as \#1 and \#2, that is moving with a velocity $v$ with respect to a rest frame $S$.
		\begin{figure}[h!]
			\centering
			\includegraphics[width=0.45\textwidth]{figures/fig_length-contraction.png}
		\end{figure}
		
		Then the rod would be at rest with respect to the moving frame $S'$. If $x_1'$ and $x_2'$ are the coordinates of either ends of the rod in the frame $S'$, then the natural or \textbf{proper length} of the rod is
		\begin{align}
			L_0&=x_2'-x_1' \label{eqn:rod:natural-length}
		\end{align}
		Now, if the corresponding coordinates of either end of the rod in the rest frame $S$ are $x_1$ and $x_2$ respectively, then the length of the rod, as measured in the frame $S$ is
		\begin{align}
			L&=x_2-x_1 \label{eqn:rod:rel-length}
		\end{align}
		
		Using Lorentz transformation, if we measure the coordinates $x_1$ and $x_2$ simultaneously, we have
		\begin{align}
			x_1'=\gamma\qty(x_1-\beta ct)&\qq{and}x_2'=\gamma\qty(x_2-\beta ct) \nonumber\\
			\implies x_1'=\gamma\qty(x_1-vt)&\qq{and}x_2'=\gamma\qty(x_2-vt) \label{eqn:x1-x2-in-Sp}
		\end{align}
		
		Inserting equations (\ref{eqn:x1-x2-in-Sp}) in equation (\ref{eqn:rod:natural-length}), we have
		\begin{align}
			L_0&=\gamma\qty(x_2-vt)-\gamma\qty(x_1-vt) \nonumber\\
				&=\gamma\qty[x_2-vt-x_1+vt] \nonumber\\
				&=\gamma\qty(x_2-x_1) \nonumber\\
			\implies L_0&=\gamma L\qq{[using eqn. (\ref{eqn:rod:rel-length})]} \nonumber\\
			\implies \boxed{L_0=\dfrac{L}{\sqrt{1-\beta^2}}}&\qq{or}\boxed{L=L_0\sqrt{1-\beta^2}} \label{eqn:length-contraction}
		\end{align}
		
		We know that
		\begin{align*}
			\gamma&\geqslant 1 \\
			\implies\dfrac{L_0}{L}&\geqslant 1 \\
			\implies L&\leqslant L_0
		\end{align*}
		
		Therefore, to a stationary observer in the frame $S$, the rod seems as if it has contracted parallel to the direction of its motion.
		
		This phenomenon of length contraction is known as the \textbf{Lorentz-Fitzgerald contraction}.
		
		\begin{figure}[h!]
			\centering
			\includegraphics[width=0.5\textwidth]{figures/fig_length-contraction-example-01.png}
		\end{figure}
		
		The faster something moves, the more squashed it appears to someone observing from a stationary frame. This happens the other way around as well.
		\begin{figure}[h!]
			\centering
			\includegraphics[width=0.5\textwidth]{figures/fig_length-contraction-example-02.png}
		\end{figure}
		
	\subsection{Time dilation}
		Consider two successive events $P$ and $Q$ that take place at the same position $x'$ in the inertial frame $S'$.
		
		Let $t_1'$ and $t_2'$ be the time instants when $P$ and $Q$ occurred, as recorded by an observer in $S'$. Then $S'$ is the rest frame for this observer.
		
		The time interval between events in the rest frame of an observer, i.e. as measured by a clock in $S'$ is known as the \textbf{proper time}. Here, the proper time is
		\begin{align*}
			\Delta\tau=t_2'-t_1'
		\end{align*}
		
		Then an observer in the frame $S$ would measure these instants as per the Lorentz transformation to be
		\begin{align*}
			ct_1=\gamma\qty(ct_1'+\beta x')&\qq{and}ct_2=\gamma\qty(ct_2'+\beta x') \\
			\implies t_1=\gamma\qty(t_1'+\dfrac{\beta}{c} x')&\qq{and}t_2=\gamma\qty(t_2'+\dfrac{\beta}{c} x') \\
			\implies t_1=\gamma\qty(t_1'+\dfrac{vx'}{c^2})&\qq{and}t_2=\gamma\qty(t_2'+\dfrac{vx'}{c^2})
		\end{align*}
		
		The time interval between these two events with respect to an observer in the frame $S$ would then be
		\begin{align*}
			\Delta t&=t_2-t_1 \\
				&=\gamma\qty(t_2'+\dfrac{vx'}{c^2})-\gamma\qty(t_1'+\dfrac{vx'}{c^2}) \\
				&=\gamma\qty[t_2'+\dfrac{vx'}{c^2}-t_1'-\dfrac{vx'}{c^2}] \\
				&=\gamma\qty(t_2'-t_1') \\
			\implies\Delta t&=\gamma\Delta\tau \\
			\implies\boxed{\Delta\tau=\Delta t\sqrt{1-\beta^2}}&\qq{or}\boxed{\Delta t=\dfrac{\Delta\tau}{\sqrt{1-\beta^2}}}
		\end{align*}
		
		We know that
		\begin{align*}
			\gamma&\geqslant 1 \\
			\implies\dfrac{\Delta t}{\Delta\tau}&\geqslant 1 \\
			\implies\Delta t&\geqslant\Delta\tau
		\end{align*}
		
		Therefore, to a stationary observer in the frame $S$, the time interval between the two events seems to increase.
		
		This phenomenon is known as \textbf{time dilation}. The faster something moves, the slower time appears to pass for it to someone observing from a stationary frame.
		
	\subsection{Relativity of simultaneity}
		\setcounter{equation}{0}
		Consider two events $P$ and $Q$ that occur at two different positions $x_1$ and $x_2$ at two different instants $t_1$ and $t_2$ respectively in the inertial frame $S$.
		
		Then, using the Lorentz transformations, we have the corresponding instants in some other inertial frame $S'$ as follows:
		\begin{align*}
			ct_1'=\gamma\qty(ct_1-\beta x_1)&\qq{and}ct_2'=\gamma\qty(ct_2-\beta x_2) \\
			\implies t_1'=\gamma\qty(t_1-\dfrac{\beta}{c}x_1)&\qq{and}t_2'=\gamma\qty(t_2-\dfrac{\beta}{c}x_2) \\
			\implies t_1'=\gamma\qty(t_1-\dfrac{vx_1}{c^2})&\qq{and}t_2'=\gamma\qty(t_2-\dfrac{vx_2}{c^2})
		\end{align*}
		
		This gives us the time interval between these two events as measured by an observer in the frame $S'$ to be
		\begin{align}
			t_2'-t_1'&=\gamma\qty(t_2-\dfrac{vx_2}{c^2})-\gamma\qty(t_1-\dfrac{vx_1}{c^2}) \nonumber\\
			\implies t_2'-t_1'&=\gamma\qty(t_2-t_1)+\dfrac{\gamma v}{c^2}\qty(x_1-x_2) \label{eqn:rel-simult-01}
		\end{align}
		
		Now, suppose both events occur simultaneously at their respective positions in the frame $S$, i.e. $t_1=t_2$. Then in equation (\ref{eqn:rel-simult-01}), we would have
		\begin{align}
			t_2'-t_1'&=\dfrac{\gamma v}{c^2}\qty(x_1-x_2)\neq 0 \label{eqn:rel-simult-02}
		\end{align}
		
		Equation (\ref{eqn:rel-simult-02}) indicates that two events, which are simultaneous in one reference frame, are not simultaneous in another moving relative to the first, unless both events occur at the same position in space.
